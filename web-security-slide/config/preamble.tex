\newcommand{\SectionDivider}[4]{%
\begin{frame}[plain]
\begin{tikzpicture}[remember picture,overlay]
  % Left color block
  \fill[#4] (current page.south west) rectangle ([xshift=0.35\paperwidth]current page.north west);
  % Direction number
  \node[anchor=west, white, font=\fontsize{72}{80}\selectfont\bfseries] at ([xshift=0.06\paperwidth, yshift=0.08\paperheight]current page.west) {#1};
  % Direction label
  \node[anchor=west, white!80, font=\scriptsize\bfseries] at ([xshift=0.06\paperwidth, yshift=-0.05\paperheight]current page.west) {DIRECTION};
  % Direction name on right
  \node[anchor=west, InkDeep, font=\LARGE\bfseries, text width=0.55\paperwidth] at ([xshift=0.40\paperwidth, yshift=0.08\paperheight]current page.west) {#2};
  % Focus description
  \node[anchor=west, SoftInk, font=\small, text width=0.50\paperwidth] at ([xshift=0.40\paperwidth, yshift=-0.02\paperheight]current page.west) {#3};
\end{tikzpicture}
\end{frame}}
